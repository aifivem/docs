```ts title="TypeScript"
import { onClientCallback, inventory, online, fail } from "@aicore-fivem/lib/server";
```

Der Server-Einstieg exportiert auch die `/fivem`-Helfer weiter: siehe [Lib: shared und fivem](/de/docs/developers/api/lib-shared). Die letzte Spalte nennt das Lua-Gegenstück auf `aiLib` ([Lib für Lua](/de/docs/developers/api/lua)).

## Client-Eingaben

Jede Client-Anfrage geht durch das Gate: Prüfung auf echten Spieler, Ratenlimit, Guards, ein Argument-Parser, optional ein Aufruf zur Zeit.

| Helfer | Was | Lua |
| --- | --- | --- |
| `onClientCallback(name, handler, options?)` / `onClientCallback(name, { args, guards?, rate?, lock? }, handler)` | Beantwortet `triggerServerCallback`; `args` macht aus rohen Werten geprüfte (null lehnt ab). | `onClientCallback` (Optionen `rate`, `lock`, `ace`) |
| `onClientEvent(name, handler, options?)` / `onClientEvent(name, { args, ... }, handler)` | Ein Net-Event von einem Client, durch dasselbe Gate. | `onClientEvent` |
| `triggerClientCallback(name, source, args = [], timeoutMs?): Promise<unknown>` | Fragt einen Client (dort `onServerCallback`). | `triggerClientCallback` |
| `isAdmin(ace?)`, `hasCharacter()`, `near(point \| (source) => point, metres)` | Guards für `guards: [...]`. | `isAdmin(source, ace?)` |
| `eventSource(): number` | Der Spieler, der das Event gesendet hat (vor dem ersten `await` lesen). | |
| `isOnline(source): boolean` | Ob der Spieler verbunden ist. | |
| `progressFor(source, { label, duration, anim?, scenario?, face?, allowVehicle?, canCancel?, freeze?, disable?, minRatio? }): Promise<boolean>` | Eine zeitgesteuerte Aktion, die der Server prüft: der Balken von ai-lib auf dem Client des Spielers, nur true, wenn er dort abgelaufen ist und hier genug Zeit vergangen ist. | `progressFor` (wartet) |
| `minigameFor(source, { game, difficulty?, rounds?, keys?, pins?, picks?, minMs? }): Promise<"won" \| "lost" \| "broken">` | Ein Skill-Check oder Schlossknacken, das ein Server-Ablauf braucht; ein Sieg schneller als `minMs` zählt als verloren. | `minigameFor` (wartet) |
| `new PendingActions({ minRatio? })`: `start(source, data, ms)`, `take(source, token)`, `peek`, `cancel` | Dialoge, auf die der Server wartet: ein Token, freigegeben beim Verlassen. | |

```ts title="TypeScript"
onClientCallback(
  "repair",
  { args: oneArg(readId), guards: [hasCharacter()], lock: true },
  async (source, netId) => {
    if (!inventory.hasItem(source, "repairkit")) return fail("NO_KIT");
    if (!(await progressFor(source, { label: t("REPAIRING"), duration: 8000 })))
      return fail("CANCELLED");
    inventory.removeItem(source, "repairkit");
    return { ok: true };
  },
);
```

```lua title="Lua"
aiLib.onClientCallback('repair', function(source, netId)
  if not aiLib.inventory.hasItem(source, 'repairkit') then return aiLib.fail('NO_KIT') end
  if not aiLib.progressFor(source, { label = 'Repairing', duration = 8000 }) then
    return aiLib.fail('CANCELLED')
  end
  aiLib.inventory.removeItem(source, 'repairkit', 1)
  return { ok = true }
end, { lock = true })
```

| `online.allows(source, rule: AccessRule): boolean` | Ob der Spieler einen Ort nutzen darf. | |

## Befehle, Panel, Daten

| Helfer | Was | Lua |
| --- | --- | --- |
| `addCommand(name \| [name, ...aliases], { help, params, restricted }, (source, args, raw) => ...)` | Ein Befehl: geparste Parameter, per ACE geprüft, im Panel gelistet. | `addCommand` |
| `new AdminPanel(schema)`: `get`, `onChange`, `view`, `records`, `onAction`, `rule` | Der Panel-Speicher eines Skripts: Schema, Validierung, Speichern, Deltas. | |
| `new DataFile(resource, name, log)` | Panel-Inhalt in `data/*.json` mit atomaren Schreibvorgängen. | |
| `createDatabaseTables(resource, file): Promise<void>` | Installiert `sql/main.sql`. | |
| `new WriteQueue(onError)`: `push(label, task)`, `idle()` | Datenbank-Schreibvorgänge nacheinander. | |
| `new AreaStream({ event, view, cellSize?, range?, chunk?, alsoTo?, extra? })` | Weltobjekte, die zu den Spielern in der Nähe gestreamt werden: ein Snapshot, dann nur Änderungen. | |
| `EntityState`, `PlayerState`, `GlobalKey`, `GlobalGroup` (mit `set`) | State Bags; die Server-Klassen schreiben. | |

```ts title="TypeScript"
addCommand(
  "heal",
  {
    help: t("CMD_HEAL"),
    params: [{ name: "id", type: "playerId", optional: true }],
    restricted: true,
  },
  (source, args) => medical.heal(args.id ?? source),
);
```
